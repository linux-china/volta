46impl Volta {
47    pub(crate) fn run(self, session: &mut Session) -> Fallible<ExitCode> {
48        if self.version {
49            // suffix indicator for dev build
50            if cfg!(debug_assertions) {
51                println!("{}-dev", env!("CARGO_PKG_VERSION"));
52            } else {
53                println!("{}", env!("CARGO_PKG_VERSION"));
54            }
55            Ok(ExitCode::Success)
56        } else if let Some(command) = self.command {
57            command.run(session)
58        } else {
59            Volta::parse_from(["volta", "help"].iter()).run(session)
60        }
61    }
62}

65pub(crate) enum Subcommand {
66    /// Fetches a tool to the local machine
67    Fetch(command::Fetch),
68
69    /// Installs a tool in your toolchain
70    Install(command::Install),
71
72    /// Uninstalls a tool from your toolchain
73    Uninstall(command::Uninstall),
74
75    /// Pins your project's runtime or package manager
76    Pin(command::Pin),
77
78    /// Displays the current toolchain
79    #[command(alias = "ls")]
80    List(command::List),
81
82    /// Generates Volta completions
83    ///
84    /// By default, completions will be generated for the value of your current shell,
85    /// shell, i.e. the value of `SHELL`. If you set the `<shell>` option, completions
86    /// will be generated for that shell instead.
87    ///
88    /// If you specify a directory, the completions will be written to a file there;
89    /// otherwise, they will be written to `stdout`.
90    #[command(arg_required_else_help = true)]
91    Completions(command::Completions),
92
93    /// Locates the actual binary that will be called by Volta
94    Which(command::Which),
95
96    #[command(long_about = crate::command::r#use::USAGE, hide = true)]
97    Use(command::Use),
98
99    /// Enables Volta for the current user / shell
100    Setup(command::Setup),
101
102    /// Run a command with custom Node, npm, pnpm, and/or Yarn versions
103    Run(command::Run),
104
105    /// Man pages
106    Man(command::ManPages),
107}
108
109impl Subcommand {
110    pub(crate) fn run(self, session: &mut Session) -> Fallible<ExitCode> {
111        match self {
112            Subcommand::Fetch(fetch) => fetch.run(session),
113            Subcommand::Install(install) => install.run(session),
114            Subcommand::Uninstall(uninstall) => uninstall.run(session),
115            Subcommand::Pin(pin) => pin.run(session),
116            Subcommand::List(list) => list.run(session),
117            Subcommand::Completions(completions) => completions.run(session),
118            Subcommand::Which(which) => which.run(session),
119            Subcommand::Use(r#use) => r#use.run(session),
120            Subcommand::Setup(setup) => setup.run(session),
121            Subcommand::Run(run) => run.run(session),
122            Subcommand::Man(man) => man.run(session),
123        }
124    }
125}
