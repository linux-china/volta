14/// The entry point for the `volta` CLI.
15pub fn main() {
16    rustls::crypto::ring::default_provider().install_default().expect("Failed to install rustls crypto provider");
17    let volta = cli::Volta::parse();
18    let verbosity = match (&volta.verbose, &volta.quiet) {
19        (false, false) => LogVerbosity::Default,
20        (true, false) => {
21            if volta.very_verbose {
22                LogVerbosity::VeryVerbose
23            } else {
24                LogVerbosity::Verbose
25            }
26        }
27        (false, true) => LogVerbosity::Quiet,
28        (true, true) => {
29            unreachable!("Clap should prevent the user from providing both --verbose and --quiet")
30        }
31    };
32    Logger::init(LogContext::Volta, verbosity).expect("Only a single logger should be initialized");
33    log::trace!("log level: {verbosity:?}");
34
35    let mut session = Session::init();
36    session.add_event_start(ActivityKind::Volta);
37
38    let result = ensure_layout().and_then(|()| volta.run(&mut session).map_err(Error::Volta));
39    match result {
40        Ok(exit_code) => {
41            session.add_event_end(ActivityKind::Volta, exit_code);
42            session.exit(exit_code);
43        }
44        Err(Error::Tool(code)) => {
45            session.add_event_tool_end(ActivityKind::Volta, code);
46            session.exit_tool(code);
47        }
48        Err(Error::Volta(err)) => {
49            report_error(env!("CARGO_PKG_VERSION"), &err);
50            session.add_event_error(ActivityKind::Volta, &err);
51            let code = err.exit_code();
52            session.add_event_end(ActivityKind::Volta, code);
53            session.exit(code);
54        }
55    }
56}
