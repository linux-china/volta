15/// Opens a file, creating it if it doesn't exist
16pub fn touch(path: &Path) -> io::Result<File> {
17    if !path.is_file() {
18        if let Some(basedir) = path.parent() {
19            create_dir_all(basedir)?;
20        }
21        File::create(path)?;
22    }
23    File::open(path)
24}

26/// Removes the target directory, if it exists. If the directory doesn't exist, that is treated as
27/// success.
28pub fn remove_dir_if_exists<P: AsRef<Path>>(path: P) -> Fallible<()> {
29    fs::remove_dir_all(&path)
30        .or_else(ok_if_not_found)
31        .with_context(|| ErrorKind::DeleteDirectoryError {
32            directory: path.as_ref().to_owned(),
33        })
34}

36/// Removes the target file, if it exists. If the file doesn't exist, that is treated as success.
37pub fn remove_file_if_exists<P: AsRef<Path>>(path: P) -> Fallible<()> {
38    fs::remove_file(&path)
39        .or_else(ok_if_not_found)
40        .with_context(|| ErrorKind::DeleteFileError {
41            file: path.as_ref().to_owned(),
42        })
43}

45/// Converts a failure because of file not found into a success.
46///
47/// Handling the error is preferred over checking if a file exists before removing it, since
48/// that avoids a potential race condition between the check and the removal.
49pub fn ok_if_not_found<T: Default>(err: io::Error) -> io::Result<T> {
50    match err.kind() {
51        io::ErrorKind::NotFound => Ok(T::default()),
52        _ => Err(err),
53    }
54}

56/// Reads a file, if it exists.
57pub fn read_file<P: AsRef<Path>>(path: P) -> io::Result<Option<String>> {
58    let result: io::Result<String> = fs::read_to_string(path);
59
60    match result {
61        Ok(string) => Ok(Some(string)),
62        Err(error) => match error.kind() {
63            io::ErrorKind::NotFound => Ok(None),
64            _ => Err(error),
65        },
66    }
67}

69/// Reads the full contents of a directory, eagerly extracting each directory entry
70/// and its metadata and returning an iterator over them. Returns `Error` if any of
71/// these steps fails.
72///
73/// This function makes it easier to write high level logic for manipulating the
74/// contents of directories (map, filter, etc).
75///
76/// Note that this function allocates an intermediate vector of directory entries to
77/// construct the iterator from, so if a directory is expected to be very large, it
78/// will allocate temporary data proportional to the number of entries.
79pub fn read_dir_eager(dir: &Path) -> io::Result<impl Iterator<Item = (DirEntry, Metadata)>> {
80    let entries = read_dir(dir)?;
81    let vec = entries
82        .map(|entry| {
83            let entry = entry?;
84            let metadata = entry.metadata()?;
85            Ok((entry, metadata))
86        })
87        .collect::<io::Result<Vec<(DirEntry, Metadata)>>>()?;
88
89    Ok(vec.into_iter())
90}

92/// Reads the contents of a directory and returns a Vec of the matched results
93/// from the input function
94pub fn dir_entry_match<T, F>(dir: &Path, mut f: F) -> io::Result<Vec<T>>
95where
96    F: FnMut(&DirEntry) -> Option<T>,
97{
98    let entries = read_dir_eager(dir)?;
99    Ok(entries
100        .filter(|(_, metadata)| metadata.is_file())
101        .filter_map(|(entry, _)| f(&entry))
102        .collect::<Vec<T>>())
103}

121/// Create a file symlink. The `dst` path will be a symbolic link pointing to the `src` path.
122pub fn symlink_file<S, D>(src: S, dest: D) -> io::Result<()>
123where
124    S: AsRef<Path>,
125    D: AsRef<Path>,
126{
127    #[cfg(windows)]
128    return std::os::windows::fs::symlink_file(src, dest);
129
130    #[cfg(unix)]
131    return std::os::unix::fs::symlink(src, dest);
132}

143    #[cfg(unix)]
144    return std::os::unix::fs::symlink(src, dest);
145}

147/// Ensure that a given file has 'executable' permissions, otherwise we won't be able to call it
148#[cfg(unix)]
149pub fn set_executable(bin: &Path) -> io::Result<()> {
150    let mut permissions = fs::metadata(bin)?.permissions();
151    let mode = permissions.mode();
152
153    if mode & 0o111 != 0o111 {
154        permissions.set_mode(mode | 0o111);
155        fs::set_permissions(bin, permissions)
156    } else {
157        Ok(())
158    }
159}

169/// Rename a file or directory to a new name, retrying if the operation fails because of permissions
170///
171/// Will retry for ~30 seconds with longer and longer delays between each, to allow for virus scan
172/// and other automated operations to complete.
173pub fn rename<F, T>(from: F, to: T) -> io::Result<()>
174where
175    F: AsRef<Path>,
176    T: AsRef<Path>,
177{
178    // 21 Fibonacci steps starting at 1 ms is ~28 seconds total
179    // See https://github.com/rust-lang/rustup/pull/1873 where this was used by Rustup to work around
180    // virus scanning file locks
181    let from = from.as_ref();
182    let to = to.as_ref();
183
184    retry(Fibonacci::from_millis(1).take(21), || {
185        match fs::rename(from, to) {
186            Ok(_) => OperationResult::Ok(()),
187            Err(e) => match e.kind() {
188                io::ErrorKind::PermissionDenied => OperationResult::Retry(e),
189                _ => OperationResult::Err(e),
190            },
191        }
192    })
193    .map_err(|e| e.error)
