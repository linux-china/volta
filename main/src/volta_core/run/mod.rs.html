23/// Environment variable set internally when a shim has been executed and the context evaluated
24///
25/// This is set when executing a shim command. If this is already, then the built-in shims (Node,
26/// npm, npx, pnpm and Yarn) will assume that the context has already been evaluated & the PATH has
27/// already been modified, so they will use the pass-through behavior.
28///
29/// Shims should only be called recursively when the environment is misconfigured, so this will
30/// prevent infinite recursion as the pass-through logic removes the shim directory from the PATH.
31///
32/// Note: This is explicitly _removed_ when calling a command through `volta run`, as that will
33/// never happen due to the Volta environment.
34const RECURSION_ENV_VAR: &str = "_VOLTA_TOOL_RECURSION";
35const VOLTA_BYPASS: &str = "VOLTA_BYPASS";
36
37/// Execute a shim command, based on the command-line arguments to the current process
38pub fn execute_shim(session: &mut Session) -> Fallible<ExitStatus> {
39    let mut native_args = env::args_os();
40    let exe = get_tool_name(&mut native_args)?;
41    let args: Vec<_> = native_args.collect();
42
43    get_executor(&exe, &args, session)?.execute(session)
44}

46/// Execute a tool with the provided arguments
47pub fn execute_tool<K, V, S>(
48    exe: &OsStr,
49    args: &[OsString],
50    envs: &HashMap<K, V, S>,
51    cli: CliPlatform,
52    session: &mut Session,
53) -> Fallible<ExitStatus>
54where
55    K: AsRef<OsStr>,
56    V: AsRef<OsStr>,
57{
58    // Remove the recursion environment variable so that the context is correctly re-evaluated
59    // when calling `volta run` (even when called from a Node script)
60    unsafe {
61        env::remove_var(RECURSION_ENV_VAR);
62    }
63
64    let mut runner = get_executor(exe, args, session)?;
65    runner.cli_platform(cli);
66    runner.envs(envs);
67
68    runner.execute(session)
69}

71/// Get the appropriate Tool command, based on the requested executable and arguments
72fn get_executor(
73    exe: &OsStr,
74    args: &[OsString],
75    session: &mut Session,
76) -> Fallible<executor::Executor> {
77    if env::var_os(VOLTA_BYPASS).is_some() {
78        Ok(executor::ToolCommand::new(
79            exe,
80            args,
81            None,
82            executor::ToolKind::Bypass(exe.to_string_lossy().to_string()),
83        )
84        .into())
85    } else {
86        match exe.to_str() {
87            Some("volta-shim") => Err(ErrorKind::RunShimDirectly.into()),
88            Some("node") => node::command(args, session),
89            Some("npm") => npm::command(args, session),
90            Some("npx") => npx::command(args, session),
91            Some("pnpm") => {
92                // If the pnpm feature flag variable is set, delegate to the pnpm handler
93                // If not, use the binary handler as a fallback (prior to pnpm support, installing
94                // pnpm would be handled the same as any other global binary)
95                if env::var_os(VOLTA_FEATURE_PNPM).is_some() {
96                    pnpm::command(args, session)
97                } else {
98                    binary::command(exe, args, session)
99                }
100            }
101            Some("yarn") | Some("yarnpkg") => yarn::command(args, session),
102            _ => binary::command(exe, args, session),
103        }
104    }
105}

107/// Determine the name of the command to run by inspecting the first argument to the active process
108fn get_tool_name(args: &mut ArgsOs) -> Fallible<OsString> {
109    args.next()
110        .and_then(|arg0| Path::new(&arg0).file_name().map(tool_name_from_file_name))
111        .ok_or_else(|| ErrorKind::CouldNotDetermineTool.into())
112}
113
114#[cfg(unix)]
115fn tool_name_from_file_name(file_name: &OsStr) -> OsString {
116    file_name.to_os_string()
117}
118
119#[cfg(windows)]
120fn tool_name_from_file_name(file_name: &OsStr) -> OsString {
121    // On Windows PowerShell, the file name includes the .exe suffix,
122    // and the Windows file system is case-insensitive
123    // We need to remove that to get the raw tool name
124    match file_name.to_str() {
125        Some(file) => OsString::from(file.to_ascii_lowercase().trim_end_matches(".exe")),
126        None => OsString::from(file_name),
127    }
128}

136/// Write a debug message with the full image that will be used to execute a command
137#[inline]
138fn debug_active_image(image: &Image) {
139    debug!(
140        "Active Image:
141    Node: {}
142    npm: {}
143    pnpm: {}
144    Yarn: {}",
145        format_tool_version(&image.node),
146        image
147            .resolve_npm()
148            .ok()
149            .as_ref()
150            .map(format_tool_version)
151            .unwrap_or_else(|| "Bundled with Node".into()),
152        image
153            .pnpm
154            .as_ref()
155            .map(format_tool_version)
156            .unwrap_or_else(|| "None".into()),
157        image
158            .yarn
159            .as_ref()
160            .map(format_tool_version)
161            .unwrap_or_else(|| "None".into()),
162    )
163}
