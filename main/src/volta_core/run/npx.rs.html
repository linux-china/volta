11const REQUIRED_NPM_VERSION: Version = Version::new(5, 2, 0);

13/// Build a `ToolCommand` for npx
14pub(super) fn command(args: &[OsString], session: &mut Session) -> Fallible<Executor> {
15    session.add_event_start(ActivityKind::Npx);
16    // Don't re-evaluate the context if this is a recursive call
17    let platform = match env::var_os(RECURSION_ENV_VAR) {
18        Some(_) => None,
19        None => Platform::current(session)?,
20    };
21
22    Ok(ToolCommand::new("npx", args, platform, ToolKind::Npx).into())
23}
24
25/// Determine the execution context (PATH and failure error message) for npx
26pub(super) fn execution_context(
27    platform: Option<Platform>,
28    session: &mut Session,
29) -> Fallible<(OsString, ErrorKind)> {
30    match platform {
31        Some(plat) => {
32            let image = plat.checkout(session)?;
33
34            // If the npm version is lower than the minimum required, we can show a helpful error
35            // message instead of a 'command not found' error.
36            let active_npm = image.resolve_npm()?;
37            if active_npm.value < REQUIRED_NPM_VERSION {
38                return Err(ErrorKind::NpxNotAvailable {
39                    version: active_npm.value.to_string(),
40                }
41                .into());
42            }
43
44            let path = image.path()?;
45            debug_active_image(&image);
46
47            Ok((path, ErrorKind::BinaryExecError))
48        }
49        None => {
50            let path = System::path()?;
51            debug_no_platform();
52            Ok((path, ErrorKind::NoPlatform))
53        }
54    }
55}
