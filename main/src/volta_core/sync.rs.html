1//! Inter-process locking on the Volta directory
2//!
3//! To avoid issues where multiple separate invocations of Volta modify the
4//! data directory simultaneously, we provide a locking mechanism that only
5//! allows a single process to modify the directory at a time.
6//!
7//! However, within a single process, we may attempt to lock the directory in
8//! different code paths. For example, when installing a package we require a
9//! lock, however we also may need to install Node, which requires a lock as
10//! well. To avoid deadlocks in those situations, we track the state of the
11//! lock globally:
12//!
13//! - If a lock is requested and no locks are active, then we acquire a file
14//!   lock on the `volta.lock` file and initialize the state with a count of 1
15//! - If a lock already exists, then we increment the count of active locks
16//! - When a lock is no longer needed, we decrement the count of active locks
17//! - When the last lock is released, we release the file lock and clear the
18//!   global lock state.
19//!
20//! This allows multiple code paths to request a lock and not worry about
21//! potential deadlocks, while still preventing multiple processes from making
22//! concurrent changes.
23
24use std::fs::{File, OpenOptions};
25use std::marker::PhantomData;
26use std::ops::Drop;
27use std::sync::Mutex;
28
29use crate::error::{Context, ErrorKind, Fallible};
30use crate::layout::volta_home;
31use crate::style::progress_spinner;
32use fs2::FileExt;
33use log::debug;
34use once_cell::sync::Lazy;
35
36static LOCK_STATE: Lazy<Mutex<Option<LockState>>> = Lazy::new(|| Mutex::new(None));
37
38/// The current state of locks for this process.
39///
40/// Note: To ensure thread safety _within_ this process, we enclose the
41/// state in a Mutex. This Mutex and it's associated locks are separate
42/// from the overall process lock and are only used to ensure the count
43/// is accurately maintained within a given process.
44struct LockState {
45    file: File,
46    count: usize,
47}
48
49const LOCK_FILE: &str = "volta.lock";
50
51/// An RAII implementation of a process lock on the Volta directory. A given Volta process can have
52/// multiple active locks, but only one process can have any locks at a time.
53///
54/// Once all of the `VoltaLock` objects go out of scope, the lock will be released to other
55/// processes.
56pub struct VoltaLock {
57    // Private field ensures that this cannot be created except for with the `acquire()` method
58    _private: PhantomData<()>,
59}
60
61impl VoltaLock {
62    pub fn acquire() -> Fallible<Self> {
63        let mut state = LOCK_STATE
64            .lock()
65            .with_context(|| ErrorKind::LockAcquireError)?;
66
67        // Check if there is an active lock for this process. If so, increment
68        // the count of active locks. If not, create a file lock and initialize
69        // the state with a count of 1
70        match &mut *state {
71            Some(inner) => {
72                inner.count += 1;
73            }
74            None => {
75                let path = volta_home()?.root().join(LOCK_FILE);
76                debug!("Acquiring lock on Volta directory: {}", path.display());
77
78                let file = OpenOptions::new()
79                    .write(true)
80                    .create(true)
81                    .open(path)
82                    .with_context(|| ErrorKind::LockAcquireError)?;
83                // First we try to lock the file without blocking. If that fails, then we show a spinner
84                // and block until the lock completes.
85                if file.try_lock_exclusive().is_err() {
86                    let spinner = progress_spinner("Waiting for file lock on Volta directory");
87                    // Note: Blocks until the file can be locked
88                    let lock_result = file
89                        .lock_exclusive()
90                        .with_context(|| ErrorKind::LockAcquireError);
91                    spinner.finish_and_clear();
92                    lock_result?;
93                }
94
95                *state = Some(LockState { file, count: 1 });
96            }
97        }
98
99        Ok(Self {
100            _private: PhantomData,
101        })
102    }
103}
104
105impl Drop for VoltaLock {
106    fn drop(&mut self) {
107        // On drop, decrement the count of active locks. If the count is 1,
108        // then this is the last active lock, so instead unlock the file and
109        // clear out the lock state.
110        if let Ok(mut state) = LOCK_STATE.lock() {
111            match &mut *state {
112                Some(inner) => {
113                    if inner.count == 1 {
114                        debug!("Unlocking Volta Directory");
115                        let _ = inner.file.unlock();
116                        *state = None;
117                    } else {
118                        inner.count -= 1;
119                    }
120                }
121                None => {
122                    debug!("Unexpected unlock of Volta directory when it wasn't locked");
123                }
124            }
125        }
126    }
127}
