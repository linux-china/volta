29/// The Tool implementation for installing 3rd-party global packages
30pub struct Package {
31    name: String,
32    version: VersionSpec,
33    staging: TempDir,
34}
35
36impl Package {
37    pub fn new(name: String, version: VersionSpec) -> Fallible<Self> {
38        let staging = setup_staging_directory(PackageManager::Npm, NeedsScope::No)?;
39
40        Ok(Package {
41            name,
42            version,
43            staging,
44        })
45    }
46
47    pub fn run_install(&self, platform_image: &Image) -> Fallible<()> {
48        install::run_global_install(
49            self.to_string(),
50            self.staging.path().to_owned(),
51            platform_image,
52        )
53    }
54
55    pub fn complete_install(self, image: &Image) -> Fallible<PackageManifest> {
56        let manager = PackageManager::Npm;
57        let manifest =
58            configure::parse_manifest(&self.name, self.staging.path().to_owned(), manager)?;
59
60        persist_install(&self.name, &self.version, self.staging.path())?;
61        link_package_to_shared_dir(&self.name, manager)?;
62        configure::write_config_and_shims(&self.name, &manifest, image, manager)?;
63
64        Ok(manifest)
65    }
66}
67
68impl Tool for Package {
69    fn fetch(self: Box<Self>, _session: &mut Session) -> Fallible<()> {
70        Err(ErrorKind::CannotFetchPackage {
71            package: self.to_string(),
72        }
73        .into())
74    }
75
76    fn install(self: Box<Self>, session: &mut Session) -> Fallible<()> {
77        let _lock = VoltaLock::acquire();
78
79        let default_image = session
80            .default_platform()?
81            .map(PlatformSpec::as_default)
82            .ok_or(ErrorKind::NoPlatform)?
83            .checkout(session)?;
84
85        self.run_install(&default_image)?;
86        let manifest = self.complete_install(&default_image)?;
87
88        let bins = manifest.bin.join(", ");
89
90        if bins.is_empty() {
91            info!(
92                "{} installed {}",
93                success_prefix(),
94                tool_version(manifest.name, manifest.version)
95            );
96        } else {
97            info!(
98                "{} installed {} with executables: {}",
99                success_prefix(),
100                tool_version(manifest.name, manifest.version),
101                bins
102            );
103        }
104
105        Ok(())
106    }
107
108    fn pin(self: Box<Self>, _session: &mut Session) -> Fallible<()> {
109        Err(ErrorKind::CannotPinPackage { package: self.name }.into())
110    }
111}
112
113impl Display for Package {
114    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
115        match self.version {
116            VersionSpec::None => f.write_str(&self.name),
117            _ => f.write_str(&tool_version(&self.name, &self.version)),
118        }
119    }
120}

122/// Helper struct for direct installs through `npm i -g` or `yarn global add`
123///
124/// Provides methods to simplify installing into a staging directory and then moving that install
125/// into the proper location after it is complete.
126///
127/// Note: We don't always know the name of the package up-front, as the install could be from a
128/// tarball or a git coordinate. If we do know ahead of time, then we can skip looking it up
129pub struct DirectInstall {
130    staging: TempDir,
131    manager: PackageManager,
132    name: Option<String>,
133}
134
135impl DirectInstall {
136    pub fn new(manager: PackageManager) -> Fallible<Self> {
137        let staging = setup_staging_directory(manager, NeedsScope::No)?;
138
139        Ok(DirectInstall {
140            staging,
141            manager,
142            name: None,
143        })
144    }
145
146    pub fn with_name(manager: PackageManager, name: String) -> Fallible<Self> {
147        let staging = setup_staging_directory(manager, name.contains('/').into())?;
148
149        Ok(DirectInstall {
150            staging,
151            manager,
152            name: Some(name),
153        })
154    }
155
156    pub fn setup_command(&self, command: &mut Command) {
157        self.manager
158            .setup_global_command(command, self.staging.path().to_owned());
159    }
160
161    pub fn complete_install(self, image: &Image) -> Fallible<()> {
162        let DirectInstall {
163            staging,
164            name,
165            manager,
166        } = self;
167
168        let mut package_name = name.or_else(|| manager.get_installed_package(staging.path().to_owned()));
169        if package_name.is_none() {
170            // extract package name(last arg) from args
171            let args: Vec<String> = std::env::args().collect();
172            if let Some(last_item) = args.last() {
173                let mut name_arg = last_item.as_str();
174                if let Some(pos) = last_item.rfind('@') {
175                    if pos != 0 {
176                        name_arg = &last_item[..pos];
177                    }
178                }
179                package_name = Some(name_arg.to_string());
180            }
181        }
182        let name = package_name.ok_or(ErrorKind::InstalledPackageNameError)?;
183        let manifest = configure::parse_manifest(&name, staging.path().to_owned(), manager)?;
184
185        persist_install(&name, &manifest.version, staging.path())?;
186        link_package_to_shared_dir(&name, manager)?;
187        configure::write_config_and_shims(&name, &manifest, image, manager)
188    }
189}

191/// Helper struct for direct in-place upgrades using `npm update -g` or `yarn global upgrade`
192///
193/// Upgrades the requested package directly in the image directory
194pub struct InPlaceUpgrade {
195    package: String,
196    directory: PathBuf,
197    manager: PackageManager,
198}
199
200impl InPlaceUpgrade {
201    pub fn new(package: String, manager: PackageManager) -> Fallible<Self> {
202        let directory = volta_home()?.package_image_dir(&package);
203
204        Ok(Self {
205            package,
206            directory,
207            manager,
208        })
209    }
210
211    /// Check for possible failure cases with the package to be upgraded
212    ///     - The package is not installed as a global
213    ///     - The package exists, but was installed with a different package manager
214    pub fn check_upgraded_package(&self) -> Fallible<()> {
215        let config =
216            PackageConfig::from_file(volta_home()?.default_package_config_file(&self.package))
217                .with_context(|| ErrorKind::UpgradePackageNotFound {
218                    package: self.package.clone(),
219                    manager: self.manager,
220                })?;
221
222        if config.manager != self.manager {
223            Err(ErrorKind::UpgradePackageWrongManager {
224                package: self.package.clone(),
225                manager: config.manager,
226            }
227            .into())
228        } else {
229            Ok(())
230        }
231    }
232
233    pub fn setup_command(&self, command: &mut Command) {
234        self.manager
235            .setup_global_command(command, self.directory.clone());
236    }
237
238    pub fn complete_upgrade(self, image: &Image) -> Fallible<()> {
239        let manifest = configure::parse_manifest(&self.package, self.directory, self.manager)?;
240
241        link_package_to_shared_dir(&self.package, self.manager)?;
242        configure::write_config_and_shims(&self.package, &manifest, image, self.manager)
243    }
244}

262/// Create the temporary staging directory we will use to install and ensure expected
263/// subdirectories exist within it
264fn setup_staging_directory(manager: PackageManager, needs_scope: NeedsScope) -> Fallible<TempDir> {
265    // Workaround to ensure relative symlinks continue to work.
266    // The final installed location of packages is:
267    //      $VOLTA_HOME/tools/image/packages/{name}/
268    // To ensure that the temp directory has the same amount of nesting, we use:
269    //      $VOLTA_HOME/tmp/image/packages/{tempdir}/
270    // This way any relative symlinks will have the same amount of nesting and will remain valid
271    // even when the directory is persisted.
272    // We also need to handle the case when the linked package has a scope, which requires another
273    // level of nesting
274    let mut staging_root = volta_home()?.tmp_dir().to_owned();
275    staging_root.push("image");
276    staging_root.push("packages");
277    if needs_scope == NeedsScope::Yes {
278        staging_root.push("scope");
279    }
280    create_dir_all(&staging_root).with_context(|| ErrorKind::ContainingDirError {
281        path: staging_root.clone(),
282    })?;
283    let staging = tempdir_in(&staging_root).with_context(|| ErrorKind::CreateTempDirError {
284        in_dir: staging_root,
285    })?;
286
287    let source_dir = manager.source_dir(staging.path().to_owned());
288    ensure_containing_dir_exists(&source_dir)
289        .with_context(|| ErrorKind::ContainingDirError { path: source_dir })?;
290
291    let binary_dir = manager.binary_dir(staging.path().to_owned());
292    ensure_containing_dir_exists(&binary_dir)
293        .with_context(|| ErrorKind::ContainingDirError { path: binary_dir })?;
294
295    Ok(staging)
296}

298fn persist_install<V>(package_name: &str, package_version: V, staging_dir: &Path) -> Fallible<()>
299where
300    V: Display,
301{
302    let package_dir = volta_home()?.package_image_dir(package_name);
303
304    remove_dir_if_exists(&package_dir)?;
305
306    // Handle scoped packages (@vue/cli), which have an extra directory for the scope
307    ensure_containing_dir_exists(&package_dir).with_context(|| ErrorKind::ContainingDirError {
308        path: package_dir.to_owned(),
309    })?;
310
311    rename(staging_dir, &package_dir).with_context(|| ErrorKind::SetupToolImageError {
312        tool: package_name.into(),
313        version: package_version.to_string(),
314        dir: package_dir,
315    })?;
316
317    Ok(())
318}

320fn link_package_to_shared_dir(package_name: &str, manager: PackageManager) -> Fallible<()> {
321    let home = volta_home()?;
322    let mut source = manager.source_dir(home.package_image_dir(package_name));
323    source.push(package_name);
324
325    let target = home.shared_lib_dir(package_name);
326
327    remove_dir_if_exists(&target)?;
328
329    // Handle scoped packages (@vue/cli), which have an extra directory for the scope
330    ensure_containing_dir_exists(&target).with_context(|| ErrorKind::ContainingDirError {
331        path: target.clone(),
332    })?;
333
334    symlink_dir(source, target).with_context(|| ErrorKind::CreateSharedLinkError {
335        name: package_name.into(),
336    })
