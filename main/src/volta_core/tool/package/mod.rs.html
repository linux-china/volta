29/// The Tool implementation for installing 3rd-party global packages
30pub struct Package {
31    name: String,
32    version: VersionSpec,
33    staging: TempDir,
34}
35
36impl Package {
37    pub fn new(name: String, version: VersionSpec) -> Fallible<Self> {
38        let staging = setup_staging_directory(PackageManager::Npm, NeedsScope::No)?;
39
40        Ok(Package {
41            name,
42            version,
43            staging,
44        })
45    }
46
47    pub fn run_install(&self, platform_image: &Image) -> Fallible<()> {
48        install::run_global_install(
49            self.to_string(),
50            self.staging.path().to_owned(),
51            platform_image,
52        )
53    }
54
55    pub fn complete_install(self, image: &Image) -> Fallible<PackageManifest> {
56        let manager = PackageManager::Npm;
57        let manifest =
58            configure::parse_manifest(&self.name, self.staging.path().to_owned(), manager)?;
59
60        // Validate before persisting, so that a rejected install leaves any existing image intact
61        configure::validate_bins(&self.name, &manifest)?;
62        persist_install(&self.name, &self.version, self.staging.path())?;
63        link_package_to_shared_dir(&self.name, manager)?;
64        configure::write_config_and_shims(&self.name, &manifest, image, manager)?;
65
66        Ok(manifest)
67    }
68}
69
70impl Tool for Package {
71    fn fetch(self: Box<Self>, _session: &mut Session) -> Fallible<()> {
72        Err(ErrorKind::CannotFetchPackage {
73            package: self.to_string(),
74        }
75        .into())
76    }
77
78    fn install(self: Box<Self>, session: &mut Session) -> Fallible<()> {
79        let _lock = VoltaLock::acquire();
80
81        let default_image = session
82            .default_platform()?
83            .map(PlatformSpec::as_default)
84            .ok_or(ErrorKind::NoPlatform)?
85            .checkout(session)?;
86
87        self.run_install(&default_image)?;
88        let manifest = self.complete_install(&default_image)?;
89
90        let bins = manifest.bin.join(", ");
91
92        if bins.is_empty() {
93            info!(
94                "{} installed {}",
95                success_prefix(),
96                tool_version(manifest.name, manifest.version)
97            );
98        } else {
99            info!(
100                "{} installed {} with executables: {}",
101                success_prefix(),
102                tool_version(manifest.name, manifest.version),
103                bins
104            );
105        }
106
107        Ok(())
108    }
109
110    fn pin(self: Box<Self>, _session: &mut Session) -> Fallible<()> {
111        Err(ErrorKind::CannotPinPackage { package: self.name }.into())
112    }
113}
114
115impl Display for Package {
116    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
117        match self.version {
118            VersionSpec::None => f.write_str(&self.name),
119            _ => f.write_str(&tool_version(&self.name, &self.version)),
120        }
121    }
122}

124/// Helper struct for direct installs through `npm i -g` or `yarn global add`
125///
126/// Provides methods to simplify installing into a staging directory and then moving that install
127/// into the proper location after it is complete.
128///
129/// Note: We don't always know the name of the package up-front, as the install could be from a
130/// tarball or a git coordinate. If we do know ahead of time, then we can skip looking it up
131pub struct DirectInstall {
132    staging: TempDir,
133    manager: PackageManager,
134    name: Option<String>,
135}
136
137impl DirectInstall {
138    pub fn new(manager: PackageManager) -> Fallible<Self> {
139        let staging = setup_staging_directory(manager, NeedsScope::No)?;
140
141        Ok(DirectInstall {
142            staging,
143            manager,
144            name: None,
145        })
146    }
147
148    pub fn with_name(manager: PackageManager, name: String) -> Fallible<Self> {
149        let staging = setup_staging_directory(manager, name.contains('/').into())?;
150
151        Ok(DirectInstall {
152            staging,
153            manager,
154            name: Some(name),
155        })
156    }
157
158    pub fn setup_command(&self, command: &mut Command) {
159        self.manager
160            .setup_global_command(command, self.staging.path().to_owned());
161    }
162
163    pub fn complete_install(self, image: &Image) -> Fallible<()> {
164        let DirectInstall {
165            staging,
166            name,
167            manager,
168        } = self;
169
170        let mut package_name = name.or_else(|| manager.get_installed_package(staging.path().to_owned()));
171        if package_name.is_none() {
172            // extract package name(last arg) from args
173            let args: Vec<String> = std::env::args().collect();
174            if let Some(last_item) = args.last() {
175                let mut name_arg = last_item.as_str();
176                if let Some(pos) = last_item.rfind('@')
177                    && pos != 0 {
178                        name_arg = &last_item[..pos];
179                    }
180                package_name = Some(name_arg.to_string());
181            }
182        }
183        let name = package_name.ok_or(ErrorKind::InstalledPackageNameError)?;
184        let manifest = configure::parse_manifest(&name, staging.path().to_owned(), manager)?;
185
186        configure::validate_bins(&name, &manifest)?;
187        persist_install(&name, &manifest.version, staging.path())?;
188        link_package_to_shared_dir(&name, manager)?;
189        configure::write_config_and_shims(&name, &manifest, image, manager)
190    }
191}

193/// Helper struct for direct in-place upgrades using `npm update -g` or `yarn global upgrade`
194///
195/// Upgrades the requested package directly in the image directory
196pub struct InPlaceUpgrade {
197    package: String,
198    directory: PathBuf,
199    manager: PackageManager,
200}
201
202impl InPlaceUpgrade {
203    pub fn new(package: String, manager: PackageManager) -> Fallible<Self> {
204        let directory = volta_home()?.package_image_dir(&package);
205
206        Ok(Self {
207            package,
208            directory,
209            manager,
210        })
211    }
212
213    /// Check for possible failure cases with the package to be upgraded
214    ///     - The package is not installed as a global
215    ///     - The package exists, but was installed with a different package manager
216    pub fn check_upgraded_package(&self) -> Fallible<()> {
217        let config =
218            PackageConfig::from_file(volta_home()?.default_package_config_file(&self.package))
219                .with_context(|| ErrorKind::UpgradePackageNotFound {
220                    package: self.package.clone(),
221                    manager: self.manager,
222                })?;
223
224        if config.manager != self.manager {
225            Err(ErrorKind::UpgradePackageWrongManager {
226                package: self.package.clone(),
227                manager: config.manager,
228            }
229            .into())
230        } else {
231            Ok(())
232        }
233    }
234
235    pub fn setup_command(&self, command: &mut Command) {
236        self.manager
237            .setup_global_command(command, self.directory.clone());
238    }
239
240    pub fn complete_upgrade(self, image: &Image) -> Fallible<()> {
241        let manifest = configure::parse_manifest(&self.package, self.directory, self.manager)?;
242
243        configure::validate_bins(&self.package, &manifest)?;
244        link_package_to_shared_dir(&self.package, self.manager)?;
245        configure::write_config_and_shims(&self.package, &manifest, image, self.manager)
246    }
247}

265/// Create the temporary staging directory we will use to install and ensure expected
266/// subdirectories exist within it
267fn setup_staging_directory(manager: PackageManager, needs_scope: NeedsScope) -> Fallible<TempDir> {
268    // Workaround to ensure relative symlinks continue to work.
269    // The final installed location of packages is:
270    //      $VOLTA_HOME/tools/image/packages/{name}/
271    // To ensure that the temp directory has the same amount of nesting, we use:
272    //      $VOLTA_HOME/tmp/image/packages/{tempdir}/
273    // This way any relative symlinks will have the same amount of nesting and will remain valid
274    // even when the directory is persisted.
275    // We also need to handle the case when the linked package has a scope, which requires another
276    // level of nesting
277    let mut staging_root = volta_home()?.tmp_dir().to_owned();
278    staging_root.push("image");
279    staging_root.push("packages");
280    if needs_scope == NeedsScope::Yes {
281        staging_root.push("scope");
282    }
283    create_dir_all(&staging_root).with_context(|| ErrorKind::ContainingDirError {
284        path: staging_root.clone(),
285    })?;
286    let staging = tempdir_in(&staging_root).with_context(|| ErrorKind::CreateTempDirError {
287        in_dir: staging_root,
288    })?;
289
290    let source_dir = manager.source_dir(staging.path().to_owned());
291    ensure_containing_dir_exists(&source_dir)
292        .with_context(|| ErrorKind::ContainingDirError { path: source_dir })?;
293
294    let binary_dir = manager.binary_dir(staging.path().to_owned());
295    ensure_containing_dir_exists(&binary_dir)
296        .with_context(|| ErrorKind::ContainingDirError { path: binary_dir })?;
297
298    Ok(staging)
299}

301fn persist_install<V>(package_name: &str, package_version: V, staging_dir: &Path) -> Fallible<()>
302where
303    V: Display,
304{
305    let package_dir = volta_home()?.package_image_dir(package_name);
306
307    remove_dir_if_exists(&package_dir)?;
308
309    // Handle scoped packages (@vue/cli), which have an extra directory for the scope
310    ensure_containing_dir_exists(&package_dir).with_context(|| ErrorKind::ContainingDirError {
311        path: package_dir.to_owned(),
312    })?;
313
314    rename(staging_dir, &package_dir).with_context(|| ErrorKind::SetupToolImageError {
315        tool: package_name.into(),
316        version: package_version.to_string(),
317        dir: package_dir,
318    })?;
319
320    Ok(())
321}

323fn link_package_to_shared_dir(package_name: &str, manager: PackageManager) -> Fallible<()> {
324    let home = volta_home()?;
325    let mut source = manager.source_dir(home.package_image_dir(package_name));
326    source.push(package_name);
327
328    let target = home.shared_lib_dir(package_name);
329
330    remove_dir_if_exists(&target)?;
331
332    // Handle scoped packages (@vue/cli), which have an extra directory for the scope
333    ensure_containing_dir_exists(&target).with_context(|| ErrorKind::ContainingDirError {
334        path: target.clone(),
335    })?;
336
337    symlink_dir(source, target).with_context(|| ErrorKind::CreateSharedLinkError {
338        name: package_name.into(),
339    })
