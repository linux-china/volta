17pub fn resolve(matching: VersionSpec, session: &mut Session) -> Fallible<Version> {
18    let hooks = session.hooks()?.yarn();
19    match matching {
20        VersionSpec::Semver(requirement) => resolve_semver(requirement, hooks),
21        VersionSpec::Exact(version) => Ok(version),
22        VersionSpec::None => resolve_tag(VersionTag::Latest, hooks),
23        VersionSpec::Tag(tag) => resolve_tag(tag, hooks),
24    }
25}

27fn resolve_tag(tag: VersionTag, hooks: Option<&YarnHooks>) -> Fallible<Version> {
28    // This triage is complicated because we need to maintain the legacy behavior of hooks
29    // First, if the tag is 'latest' and we have a 'latest' hook, we use the old behavior
30    // Next, if the tag is 'latest' and we _do not_ have a 'latest' hook, we use the new behavior
31    // Next, if the tag is _not_ 'latest' and we have an 'index' hook, we show an error since
32    //     the previous behavior did not support generic tags
33    // Finally, we don't have any relevant hooks, so we can use the new behavior
34    match (tag, hooks) {
35        (
36            VersionTag::Latest,
37            Some(&YarnHooks {
38                latest: Some(ref hook),
39                ..
40            }),
41        ) => {
42            debug!("Using yarn.latest hook to determine latest-version URL");
43            // does yarn3 use latest-version? no
44            resolve_latest_legacy(hook.resolve("latest-version")?)
45        }
46        (VersionTag::Latest, _) => resolve_custom_tag(VersionTag::Latest.to_string()),
47        (tag, Some(&YarnHooks { index: Some(_), .. })) => Err(ErrorKind::YarnVersionNotFound {
48            matching: tag.to_string(),
49        }
50        .into()),
51        (tag, _) => resolve_custom_tag(tag.to_string()),
52    }
53}

55fn resolve_semver(matching: Range, hooks: Option<&YarnHooks>) -> Fallible<Version> {
56    // For semver, the triage is less complicated: The previous behavior _always_ used
57    // the 'index' hook, so we can check for that to decide which behavior to use.
58    //
59    // If the user specifies a format for the registry, we use that. Otherwise Github format
60    // is the default legacy behavior.
61    if let Some(&YarnHooks {
62        index: Some(ref hook),
63        ..
64    }) = hooks
65    {
66        debug!("Using yarn.index hook to determine yarn index URL");
67        match hook.format {
68            RegistryFormat::Github => resolve_semver_legacy(matching, hook.resolve("releases")?),
69            RegistryFormat::Npm => resolve_semver_npm(matching, hook.resolve("")?),
70        }
71    } else {
72        resolve_semver_from_registry(matching)
73    }
74}

76fn fetch_yarn_index(package: &str) -> Fallible<(String, PackageIndex)> {
77    let url = public_registry_index(package);
78    fetch_npm_registry(url, "Yarn")
79}

81fn resolve_custom_tag(tag: String) -> Fallible<Version> {
82    // first try yarn2+, which uses "@yarnpkg/cli-dist" instead of "yarn"
83    if let Ok((url, mut index)) = fetch_yarn_index("@yarnpkg/cli-dist") {
84        if let Some(version) = index.tags.remove(&tag) {
85            debug!("Found yarn@{} matching tag '{}' from {}", version, tag, url);
86            if version.major == 2 {
87                return Err(ErrorKind::Yarn2NotSupported.into());
88            }
89            return Ok(version);
90        }
91    }
92    debug!(
93        "Did not find yarn matching tag '{}' from @yarnpkg/cli-dist",
94        tag
95    );
96
97    let (url, mut index) = fetch_yarn_index("yarn")?;
98    match index.tags.remove(&tag) {
99        Some(version) => {
100            debug!("Found yarn@{} matching tag '{}' from {}", version, tag, url);
101            Ok(version)
102        }
103        None => Err(ErrorKind::YarnVersionNotFound { matching: tag }.into()),
104    }
105}

107fn resolve_latest_legacy(url: String) -> Fallible<Version> {
108    let response_text = attohttpc::get(&url)
109        .send()
110        .and_then(Response::error_for_status)
111        .and_then(Response::text)
112        .with_context(|| ErrorKind::YarnLatestFetchError {
113            from_url: url.clone(),
114        })?;
115
116    debug!("Found yarn latest version ({}) from {}", response_text, url);
117    parse_version(response_text)
118}

120fn resolve_semver_from_registry(matching: Range) -> Fallible<Version> {
121    // first try yarn2+, which uses "@yarnpkg/cli-dist" instead of "yarn"
122    if let Ok((url, index)) = fetch_yarn_index("@yarnpkg/cli-dist") {
123        let matching_entries: Vec<PackageDetails> = index
124            .entries
125            .into_iter()
126            .filter(|PackageDetails { version, .. }| matching.satisfies(version))
127            .collect();
128
129        if !matching_entries.is_empty() {
130            let details_opt = matching_entries
131                .iter()
132                .find(|PackageDetails { version, .. }| version.major >= 3);
133
134            match details_opt {
135                Some(details) => {
136                    debug!(
137                        "Found yarn@{} matching requirement '{}' from {}",
138                        details.version, matching, url
139                    );
140                    return Ok(details.version.clone());
141                }
142                None => {
143                    return Err(ErrorKind::Yarn2NotSupported.into());
144                }
145            }
146        }
147    }
148    debug!(
149        "Did not find yarn matching requirement '{}' for @yarnpkg/cli-dist",
150        matching
151    );
152
153    let (url, index) = fetch_yarn_index("yarn")?;
154
155    let details_opt = index
156        .entries
157        .into_iter()
158        .find(|PackageDetails { version, .. }| matching.satisfies(version));
159
160    match details_opt {
161        Some(details) => {
162            debug!(
163                "Found yarn@{} matching requirement '{}' from {}",
164                details.version, matching, url
165            );
166            Ok(details.version)
167        }
168        // at this point Yarn is not found in either registry
169        None => Err(ErrorKind::YarnVersionNotFound {
170            matching: matching.to_string(),
171        }
172        .into()),
173    }
174}

176fn resolve_semver_legacy(matching: Range, url: String) -> Fallible<Version> {
177    let spinner = progress_spinner(format!("Fetching registry: {}", url));
178    let releases: RawYarnIndex = attohttpc::get(&url)
179        .send()
180        .and_then(Response::error_for_status)
181        .and_then(Response::json)
182        .with_context(registry_fetch_error("Yarn", &url))?;
183    let index = YarnIndex::from(releases);
184    let releases = index.entries;
185    spinner.finish_and_clear();
186    let version_opt = releases.into_iter().rev().find(|v| matching.satisfies(v));
187
188    match version_opt {
189        Some(version) => {
190            debug!(
191                "Found yarn@{} matching requirement '{}' from {}",
192                version, matching, url
193            );
194            Ok(version)
195        }
196        None => Err(ErrorKind::YarnVersionNotFound {
197            matching: matching.to_string(),
198        }
199        .into()),
200    }
201}
