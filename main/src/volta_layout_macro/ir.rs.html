1// The `proc_macro2` crate is a polyfill for advanced functionality of Rust's
2// procedural macros, not all of which have shipped in stable Rust. It's used by
3// the `syn` and `quote` crates to produce a shimmed version of the standard
4// `TokenStream` type. So internally that's the type we have to use for the
5// implementation of our macro. The actual front-end for the macro takes this
6// shimmed `TokenStream` type and converts it to the built-in `TokenStream` type
7// required by the Rust macro system.
8use proc_macro2::TokenStream;
9use quote::quote;
10use syn::{Attribute, Ident, LitStr, Visibility};
11
12// These seem to be leaked implementation details of the `quote` macro that have
13// to be imported by users. You can ignore them; they simply pacify the compiler.
14#[allow(unused_imports)]
15use quote::{pounded_var_names, quote_each_token, quote_spanned};

17/// The intermediate representation (IR) of a struct type defined by the `layout!`
18/// macro, which contains the flattened directory entries, organized into three
19/// categories:
20///
21/// - Directories
22/// - Executable files
23/// - Other files
24pub(crate) struct Ir {
25    pub(crate) name: Ident,
26    pub(crate) attrs: Vec<Attribute>,
27    pub(crate) visibility: Visibility,
28    pub(crate) dirs: Vec<Entry>,
29    pub(crate) files: Vec<Entry>,
30    pub(crate) exes: Vec<Entry>,
31}
32
33impl Ir {
34    fn dir_names(&self) -> impl Iterator<Item = &Ident> {
35        self.dirs.iter().map(|entry| &entry.name)
36    }
37
38    fn file_names(&self) -> impl Iterator<Item = &Ident> {
39        self.files.iter().map(|entry| &entry.name)
40    }
41
42    fn exe_names(&self) -> impl Iterator<Item = &Ident> {
43        self.exes.iter().map(|entry| &entry.name)
44    }
45
46    fn field_names(&self) -> impl Iterator<Item = &Ident> {
47        let dir_names = self.dir_names();
48        let file_names = self.file_names();
49        let exe_names = self.exe_names();
50        dir_names.chain(file_names).chain(exe_names)
51    }
52
53    fn to_struct_decl(&self) -> TokenStream {
54        let name = &self.name;
55
56        let attrs = self.attrs.iter();
57        let visibility = self.visibility.clone();
58
59        let field_names = self.field_names().map(|field_name| {
60            // Use the field name's span for good duplicate-field-name error messages.
61            quote_spanned! {field_name.span()=>
62                #field_name : ::std::path::PathBuf ,
63            }
64        });
65
66        quote! {
67            #(#attrs)* #visibility struct #name {
68                #(#field_names)*
69                root: ::std::path::PathBuf,
70            }
71        }
72    }
73
74    fn to_create_method(&self) -> TokenStream {
75        let name = &self.name;
76        let dir_names = self.dir_names();
77
78        quote! {
79            impl #name {
80                /// Creates all subdirectories in this directory layout.
81                pub fn create(&self) -> ::std::io::Result<()> {
82                    #(::std::fs::create_dir_all(self.#dir_names())?;)*
83                    ::std::result::Result::Ok(())
84                }
85            }
86        }
87    }
88
89    fn to_item_methods(&self) -> TokenStream {
90        let name = &self.name;
91
92        let methods = self.field_names().map(|field_name| {
93            // Markdown-formatted field name for the doc comment.
94            let markdown_field_name = format!("`{}`", field_name);
95            let markdown_field_name = LitStr::new(&markdown_field_name, field_name.span());
96
97            // Use the field name's span for good duplicate-method-name error messages.
98            quote_spanned! {field_name.span()=>
99                #[doc = "Returns the "]
100                #[doc = #markdown_field_name]
101                #[doc = " path."]
102                pub fn #field_name(&self) -> &::std::path::Path { &self.#field_name }
103            }
104        });
105
106        quote! {
107            impl #name {
108                #(#methods)*
109
110                 /// Returns the root path for this directory layout.
111                pub fn root(&self) -> &::std::path::Path { &self.root }
112            }
113        }
114    }
115
116    fn to_ctor(&self) -> TokenStream {
117        let name = &self.name;
118        let root = Ident::new("root", self.name.span());
119
120        let dir_names = self.dir_names();
121        let dir_inits = self.dirs.iter().map(|entry| entry.to_normal_init(&root));
122
123        let file_names = self.file_names();
124        let file_inits = self.files.iter().map(|entry| entry.to_normal_init(&root));
125
126        let exe_names = self.exe_names();
127        let exe_inits = self.exes.iter().map(|entry| entry.to_exe_init(&root));
128
129        let all_names = dir_names.chain(file_names).chain(exe_names);
130        let all_inits = dir_inits.chain(file_inits).chain(exe_inits);
131
132        let markdown_struct_name = format!("`{}`", name);
133        let markdown_struct_name = LitStr::new(&markdown_struct_name, name.span());
134
135        quote! {
136            impl #name {
137                #[doc = "Constructs a new instance of the "]
138                #[doc = #markdown_struct_name]
139                #[doc = " layout, rooted at `root`."]
140                pub fn new(#root: ::std::path::PathBuf) -> Self {
141                    Self {
142                        #(#all_names: #all_inits),* ,
143                        #root: #root
144                    }
145                }
146            }
147        }
148    }
149
150    pub(crate) fn codegen(&self) -> TokenStream {
151        let struct_decl = self.to_struct_decl();
152        let ctor = self.to_ctor();
153        let item_methods = self.to_item_methods();
154        let create_method = self.to_create_method();
155
156        quote! {
157            #struct_decl
158            #ctor
159            #item_methods
160            #create_method
161        }
162    }
163}

165pub(crate) struct Entry {
166    pub(crate) name: Ident,
167    pub(crate) context: Vec<LitStr>,
168    pub(crate) filename: LitStr,
169}
170
171impl Entry {
172    fn to_normal_init(&self, root: &Ident) -> TokenStream {
173        let name = &self.name;
174        let path_items = self.context.iter();
175        let name_replicated = self.context.iter().map(|_| name);
176        let filename = &self.filename;
177
178        quote! {
179            {
180                let mut #name = #root.clone();
181                #(#name_replicated.push(#path_items);)*
182                #name.push(#filename);
183                #name
184            }
185        }
186    }
187
188    fn to_exe_init(&self, root: &Ident) -> TokenStream {
189        let name = &self.name;
190        let path_items = self.context.iter();
191        let name_replicated = self.context.iter().map(|_| name);
192        let filename = &self.filename;
193
194        quote! {
195            {
196                let mut #name = #root.clone();
197                #(#name_replicated.push(#path_items);)*
198                #name.push(::std::format!("{}{}", #filename, ::std::env::consts::EXE_SUFFIX));
199                #name
200            }
201        }
202    }
