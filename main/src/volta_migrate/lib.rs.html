1//! Provides types for modeling the current state of the Volta directory and for migrating between versions
2//!
3//! A new layout should be represented by its own struct (as in the existing v0 or v1 modules)
4//! Migrations between types should be represented by `TryFrom` implementations between the layout types
5//! (see v1.rs for examples)
6//!
7//! NOTE: Since the layout file is written once the migration is complete, all migration implementations
8//! need to be aware that they may be partially applied (if something fails in the process) and should be
9//! able to re-start gracefully from an interrupted migration
10

34/// Represents the state of the Volta directory at every point in the migration process
35///
36/// Migrations should be applied sequentially, migrating from V0 to V1 to ... as needed, cycling
37/// through the possible MigrationState values.
38enum MigrationState {
39    Empty(empty::Empty),
40    V0(Box<V0>),
41    V1(Box<V1>),
42    V2(Box<V2>),
43    V3(Box<V3>),
44    V4(#[allow(dead_code)]Box<V4>),
45}

47/// Macro to simplify the boilerplate associated with detecting a tagged state.
48///
49/// Should be passed a series of tuples, each of which contains (in this order):
50///
51/// * The layout version (module name from `volta_layout` crate, e.g. `v1`)
52/// * The `MigrationState` variant name (e.g. `V1`)
53/// * The migration object itself (e.g. `V1` from the v1 module in _this_ crate)
54///
55/// The tuples should be in reverse chronological order, so that the newest is first, e.g.:
56///
57/// detect_tagged!((v3, V3, V3), (v2, V2, V2), (v1, V1, V1));
58macro_rules! detect_tagged {
59    ($(($layout:ident, $variant:ident, $migration:ident)),*) => {
60        impl MigrationState {
61            fn detect_tagged_state(home: &::std::path::Path) -> Option<Self> {
62                None
63                $(
64                    .or_else(|| detect::$layout(home))
65                )*
66            }
67        }
68
69        mod detect {
70            $(
71                pub(super) fn $layout(home: &::std::path::Path) -> Option<super::MigrationState> {
72                    let volta_home = volta_layout::$layout::VoltaHome::new(home.to_owned());
73                    if volta_home.layout_file().exists() {
74                        Some(super::MigrationState::$variant(Box::new(super::$migration::new(home.to_owned()))))
75                    } else {
76                        None
77                    }
78                }
79            )*
80        }
81    }
82}
83
84detect_tagged!((v4, V4, V4), (v3, V3, V3), (v2, V2, V2), (v1, V1, V1));

86impl MigrationState {
87    fn current() -> Fallible<Self> {
88        // First look for a tagged version (V1+). If that can't be found, then go through the triage
89        // for detecting a legacy version
90
91        let home = volta_home()?;
92
93        match MigrationState::detect_tagged_state(home.root()) {
94            Some(state) => Ok(state),
95            None => MigrationState::detect_legacy_state(home.root()),
96        }
97    }
98
99    #[allow(clippy::unnecessary_wraps)] // Needs to be Fallible for Unix
100    fn detect_legacy_state(home: &Path) -> Fallible<Self> {
101        /*
102        Triage for determining the legacy layout version:
103        - Does Volta Home exist?
104            - If yes (Windows) then V0
105            - If yes (Unix) then check if Volta Install is outside shim_dir?
106                - If yes, then V0
107                - If no, then check if $VOLTA_HOME/load.sh exists? If yes then V0
108        - Else Empty
109
110        The extra logic on Unix is necessary because Unix installs can be either inside or outside $VOLTA_HOME
111        If it is inside, then the directory necessarily must exist, so we can't use that as a determination.
112        If it is outside (and for Windows which is always outside), then if $VOLTA_HOME exists, it must be from a
113        previous, V0 installation.
114        */
115
116        let volta_home = home.to_owned();
117
118        if volta_home.exists() {
119            #[cfg(windows)]
120            return Ok(MigrationState::V0(Box::new(V0::new(volta_home))));
121
122            #[cfg(unix)]
123            {
124                let install = volta_install()?;
125                if install.root().starts_with(&volta_home) {
126                    // Installed inside $VOLTA_HOME, so need to look for `load.sh` as a marker
127                    if volta_home.join("load.sh").exists() {
128                        return Ok(MigrationState::V0(Box::new(V0::new(volta_home))));
129                    }
130                } else {
131                    // Installed outside of $VOLTA_HOME, so it must exist from a previous V0 install
132                    return Ok(MigrationState::V0(Box::new(V0::new(volta_home))));
133                }
134            }
135        }
136
137        Ok(MigrationState::Empty(empty::Empty::new(volta_home)))
138    }
139}

141pub fn run_migration() -> Fallible<()> {
142    // Acquire an exclusive lock on the Volta directory, to ensure that no other migrations are running.
143    // If this fails, however, we still need to run the migration
144    match VoltaLock::acquire() {
145        Ok(_lock) => {
146            // The lock was acquired, so we can be confident that no other migrations are running
147            detect_and_migrate()
148        }
149        Err(_) => {
150            debug!("Unable to acquire lock on Volta directory! Running migration anyway.");
151            detect_and_migrate()
152        }
153    }
154}
155
156fn detect_and_migrate() -> Fallible<()> {
157    info!("Updating your Volta directory. This may take a few moments...");
158    let mut state = MigrationState::current()?;
159
160    // To keep the complexity of writing a new migration from continuously increasing, each new
161    // layout version only needs to implement a migration from 2 states: Empty and the previously
162    // latest version. We then apply the migrations sequentially here: V0 -> V1 -> ... -> VX
163    loop {
164        state = match state {
165            MigrationState::Empty(e) => MigrationState::V3(Box::new(e.try_into()?)),
166            MigrationState::V0(zero) => MigrationState::V1(Box::new((*zero).try_into()?)),
167            MigrationState::V1(one) => MigrationState::V2(Box::new((*one).try_into()?)),
168            MigrationState::V2(two) => MigrationState::V3(Box::new((*two).try_into()?)),
169            MigrationState::V3(three) => MigrationState::V4(Box::new((*three).try_into()?)),
170            MigrationState::V4(_) => {
171                break;
172            }
173        };
174    }
175
176    regenerate_shims_for_dir(volta_home()?.shim_dir())?;
177
178    Ok(())
