17/// Represents a V2 Volta Layout (used by Volta v0.7.3 and above)
18///
19/// Holds a reference to the V2 layout struct to support potential future migrations
20pub struct V2 {
21    pub home: v2::VoltaHome,
22}
23
24impl V2 {
25    pub fn new(home: PathBuf) -> Self {
26        V2 {
27            home: v2::VoltaHome::new(home),
28        }
29    }
30
31    /// Write the layout file to mark migration to V2 as complete
32    ///
33    /// Should only be called once all other migration steps are finished, so that we don't
34    /// accidentally mark an incomplete migration as completed
35    fn complete_migration(home: v2::VoltaHome) -> Fallible<Self> {
36        debug!("Writing layout marker file");
37        File::create(home.layout_file()).with_context(|| ErrorKind::CreateLayoutFileError {
38            file: home.layout_file().to_owned(),
39        })?;
40
41        Ok(V2 { home })
42    }
43}
44
45impl TryFrom<Empty> for V2 {
46    type Error = VoltaError;
47
48    fn try_from(old: Empty) -> Fallible<V2> {
49        debug!("New Volta installation detected, creating fresh layout");
50
51        let home = v2::VoltaHome::new(old.home);
52        home.create().with_context(|| ErrorKind::CreateDirError {
53            dir: home.root().to_owned(),
54        })?;
55
56        V2::complete_migration(home)
57    }
58}
59
60impl TryFrom<V1> for V2 {
61    type Error = VoltaError;
62
63    fn try_from(old: V1) -> Fallible<V2> {
64        debug!("Migrating from V1 layout");
65
66        let new_home = v2::VoltaHome::new(old.home.root().to_owned());
67        new_home
68            .create()
69            .with_context(|| ErrorKind::CreateDirError {
70                dir: new_home.root().to_owned(),
71            })?;
72
73        // Perform the core of the migration
74        clear_default_npm(old.home.default_platform_file())?;
75        shift_node_images(&old.home, &new_home)?;
76
77        // Complete the migration, writing the V2 layout file
78        let layout = V2::complete_migration(new_home)?;
79
80        // Remove the V1 layout file, since we're now on V2 (do this after writing the V2 so that we know the migration succeeded)
81        let old_layout_file = old.home.layout_file();
82        remove_file_if_exists(old_layout_file)?;
83        Ok(layout)
84    }
85}

87/// Clear npm from the default `platform.json` file if it is set to the same value as that bundled with Node
88///
89/// This will ensure that we don't treat the default npm from a prior version of Volta as a "custom" npm that
90/// the user explicitly requested
91fn clear_default_npm(platform_file: &Path) -> Fallible<()> {
92    let platform_json = match read_to_string(platform_file) {
93        Ok(json) => json,
94        Err(error) => {
95            if error.kind() == io::ErrorKind::NotFound {
96                return Ok(());
97            } else {
98                return Err(VoltaError::from_source(
99                    error,
100                    ErrorKind::ReadPlatformError {
101                        file: platform_file.to_path_buf(),
102                    },
103                ));
104            }
105        }
106    };
107    let mut existing_platform = Platform::try_from(platform_json)?;
108
109    if let Some(node_version) = &mut existing_platform.node {
110        if let Some(npm) = &node_version.npm {
111            if let Ok(default_npm) = load_default_npm_version(&node_version.runtime) {
112                if *npm == default_npm {
113                    node_version.npm = None;
114                    write(platform_file, existing_platform.into_json()?).with_context(|| {
115                        ErrorKind::WritePlatformError {
116                            file: platform_file.to_owned(),
117                        }
118                    })?;
119                }
120            }
121        }
122    }
123
124    Ok(())
125}

127/// Move all Node images up one directory, removing the default npm version directory
128///
129/// In the V1 layout, we kept all node images in /<node_version>/<npm_version>/, however we will be
130/// storing custom npm versions in a separate image directory, so there is no need to maintain the
131/// bundled npm version in the file structure any more. This also will make it slightly easier to access
132/// the Node image, as we no longer will need to look up the bundled npm version every time.
133fn shift_node_images(old_home: &v1::VoltaHome, new_home: &v2::VoltaHome) -> Fallible<()> {
134    let temp_dir =
135        tempdir_in(new_home.tmp_dir()).with_context(|| ErrorKind::CreateTempDirError {
136            in_dir: new_home.tmp_dir().to_owned(),
137        })?;
138    let node_installs = read_dir_eager(old_home.node_image_root_dir())
139        .with_context(|| ErrorKind::ReadDirError {
140            dir: old_home.node_image_root_dir().to_owned(),
141        })?
142        .filter_map(|(entry, metadata)| {
143            if metadata.is_dir() {
144                parse_version(entry.file_name().to_string_lossy()).ok()
145            } else {
146                None
147            }
148        });
149
150    for node_version in node_installs {
151        remove_npm_version_from_node_image_dir(old_home, new_home, node_version, temp_dir.path())?;
152    }
153
154    Ok(())
155}

157/// Move a single node image up a directory, if it currently has the npm version in its path
158fn remove_npm_version_from_node_image_dir(
159    old_home: &v1::VoltaHome,
160    new_home: &v2::VoltaHome,
161    node_version: Version,
162    temp_dir: &Path,
163) -> Fallible<()> {
164    let node_string = node_version.to_string();
165    let npm_version = load_default_npm_version(&node_version)?;
166    let old_install = old_home.node_image_dir(&node_string, &npm_version.to_string());
167
168    if old_install.exists() {
169        let temp_image = temp_dir.join(&node_string);
170        let new_install = new_home.node_image_dir(&node_string);
171        rename(&old_install, &temp_image).with_context(|| ErrorKind::SetupToolImageError {
172            tool: "Node".into(),
173            version: node_string.clone(),
174            dir: temp_image.clone(),
175        })?;
176        remove_dir_if_exists(&new_install)?;
177        rename(&temp_image, &new_install).with_context(|| ErrorKind::SetupToolImageError {
178            tool: "Node".into(),
179            version: node_string,
180            dir: temp_image,
181        })?;
182    }
183    Ok(())
