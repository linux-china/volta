20/// Represents a V3 Volta layout (used by Volta v0.9.0 and above)
21///
22/// Holds a reference to the V3 layout struct to support future migrations
23pub struct V3 {
24    pub home: v3::VoltaHome,
25}
26
27impl V3 {
28    pub fn new(home: PathBuf) -> Self {
29        V3 {
30            home: v3::VoltaHome::new(home),
31        }
32    }
33
34    /// Write the layout file to mark migration to V2 as complete
35    ///
36    /// Should only be called once all other migration steps are finished, so that we don't
37    /// accidentally mark an incomplete migration as completed
38    fn complete_migration(home: v3::VoltaHome) -> Fallible<Self> {
39        debug!("Writing layout marker file");
40        File::create(home.layout_file()).with_context(|| ErrorKind::CreateLayoutFileError {
41            file: home.layout_file().to_owned(),
42        })?;
43
44        Ok(V3 { home })
45    }
46}
47
48impl TryFrom<Empty> for V3 {
49    type Error = VoltaError;
50
51    fn try_from(old: Empty) -> Fallible<Self> {
52        debug!("New Volta installation detected, creating fresh layout");
53
54        let home = v3::VoltaHome::new(old.home);
55        home.create().with_context(|| ErrorKind::CreateDirError {
56            dir: home.root().to_owned(),
57        })?;
58
59        V3::complete_migration(home)
60    }
61}
62
63impl TryFrom<V2> for V3 {
64    type Error = VoltaError;
65
66    fn try_from(old: V2) -> Fallible<Self> {
67        debug!("Migrating from V2 layout");
68
69        let new_home = v3::VoltaHome::new(old.home.root().to_owned());
70        new_home
71            .create()
72            .with_context(|| ErrorKind::CreateDirError {
73                dir: new_home.root().to_owned(),
74            })?;
75
76        // Migrate installed packages to the new workflow
77        migrate_packages(&old.home)?;
78
79        // Remove the package inventory directory, as we no longer cache package tarballs
80        remove_dir_if_exists(old.home.package_inventory_dir())?;
81
82        // Complete the migration, writing the V3 layout file
83        let layout = V3::complete_migration(new_home)?;
84
85        // Remove the V2 layout file, since we're now on V3 (do this after writing the V3 file so that we know the migration succeeded)
86        remove_file_if_exists(old.home.layout_file())?;
87
88        Ok(layout)
89    }
90}

92fn migrate_packages(old_home: &v2::VoltaHome) -> Fallible<()> {
93    let packages = get_installed_packages(old_home);
94    let mut session = Session::init();
95
96    for package in packages {
97        migrate_single_package(package, &mut session)?;
98    }
99
100    Ok(())
101}

103/// Determine a list of all installed packages that are using the legacy package config
104fn get_installed_packages(old_home: &v2::VoltaHome) -> Vec<LegacyPackageConfig> {
105    WalkDir::new(old_home.default_package_dir())
106        .max_depth(2)
107        .into_iter()
108        .filter_map(|res| match res {
109            Ok(entry) => {
110                if entry.file_type().is_file() {
111                    let config = LegacyPackageConfig::from_file(entry.path());
112
113                    // If unable to parse the config file and this isn't an already-migrated
114                    // package, then show debug information and a warning for the user.
115                    if config.is_none() && !is_migrated_config(entry.path()) {
116                        debug!("Unable to parse config file: {}", entry.path().display());
117                        if let Some(name) = entry.path().file_stem() {
118                            let name = name.to_string_lossy();
119                            warn!(
120                                "Could not migrate {}. Please run `volta install {0}` to migrate the package manually.",
121                                name
122                            );
123                        }
124                    }
125
126                    config
127                } else {
128                    None
129                }
130            }
131            Err(error) => {
132                debug!("Error reading directory entry: {}", error);
133                None
134            }
135        })
136        .collect()
137}

139/// Determine if a package has already been migrated by attempting to read the V3 PackageConfig
140fn is_migrated_config(config_path: &Path) -> bool {
141    PackageConfig::from_file(config_path).is_ok()
142}

144/// Migrate a single package to the new workflow
145///
146/// Note: This relies on the package install logic in `volta_core`. If that logic changes, then
147/// the end result may not be a valid V3 layout any more, and this migration will need to be
148/// updated. Specifically, the invariants we rely on are:
149///
150/// - Package image directory is in the same location
151/// - Package config files are in the same location and the same format
152/// - Binary config files are in the same location and the same format
153///
154/// If any of those are violated, this migration may be invalid and need to be reworked / scrapped
155fn migrate_single_package(config: LegacyPackageConfig, session: &mut Session) -> Fallible<()> {
156    let tool = Package::new(config.name, VersionSpec::Exact(config.version))?;
157
158    let platform: PlatformSpec = config.platform.into();
159    let image = platform.as_binary().checkout(session)?;
160
161    // Run the global install command
162    tool.run_install(&image)?;
163    // Overwrite the config files and image directory
164    tool.complete_install(&image)?;
165
166    Ok(())
167}
