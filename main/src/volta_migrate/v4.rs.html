13/// Represents a V4 Volta Layout (used by Volta v2.0.0 and above)
14///
15/// Holds a reference to the V4 layout struct to support potential future migrations
16pub struct V4 {
17    #[allow(dead_code)]
18    pub home: v4::VoltaHome,
19}
20
21impl V4 {
22    pub fn new(home: PathBuf) -> Self {
23        V4 {
24            home: v4::VoltaHome::new(home),
25        }
26    }
27
28    /// Write the layout file to mark migration to V4 as complete
29    ///
30    /// Should only be called once all other migration steps are finished, so that we don't
31    /// accidentally mark an incomplete migration as completed
32    fn complete_migration(home: v4::VoltaHome) -> Fallible<Self> {
33        debug!("Writing layout marker file");
34        File::create(home.layout_file()).with_context(|| ErrorKind::CreateLayoutFileError {
35            file: home.layout_file().to_owned(),
36        })?;
37
38        Ok(V4 { home })
39    }
40}
41
42impl TryFrom<Empty> for V4 {
43    type Error = VoltaError;
44
45    fn try_from(old: Empty) -> Fallible<V4> {
46        debug!("New Volta installation detected, creating fresh layout");
47
48        let home = v4::VoltaHome::new(old.home);
49        home.create().with_context(|| ErrorKind::CreateDirError {
50            dir: home.root().to_owned(),
51        })?;
52
53        V4::complete_migration(home)
54    }
55}
56
57impl TryFrom<V3> for V4 {
58    type Error = VoltaError;
59
60    fn try_from(old: V3) -> Fallible<V4> {
61        debug!("Migrating from V3 layout");
62
63        let new_home = v4::VoltaHome::new(old.home.root().to_owned());
64        new_home
65            .create()
66            .with_context(|| ErrorKind::CreateDirError {
67                dir: new_home.root().to_owned(),
68            })?;
69
70        // Perform the core of the migration
71        #[cfg(windows)]
72        {
73            migrate_shims(&new_home)?;
74            migrate_shared_directory(&new_home)?;
75        }
76
77        // Complete the migration, writing the V4 layout file
78        let layout = V4::complete_migration(new_home)?;
79
80        // Remove the V3 layout file, since we're now on V4 (do this after writing the V4 so that we know the migration succeeded)
81        let old_layout_file = old.home.layout_file();
82        remove_file_if_exists(old_layout_file)?;
83        Ok(layout)
84    }
85}

87/// Migrate Windows shims to use the new non-symlink approach. Previously, shims were created in
88/// the same way as on Unix: With symlinks to the `volta-shim` executable. Now, we use scripts that
89/// call `volta run` to execute the underlying tool. This allows us to avoid needing developer
90/// mode, making Volta more broadly usable for Windows devs.
91///
92/// To migrate the shims, we read the shim directory looking for symlinks, remove those, and then
93/// file stem (name without extension) to generate new shims.
94#[cfg(windows)]
95fn migrate_shims(new_home: &v4::VoltaHome) -> Fallible<()> {
96    use std::ffi::OsStr;
97
98    let entries = read_dir_eager(new_home.shim_dir()).with_context(|| ErrorKind::ReadDirError {
99        dir: new_home.shim_dir().to_owned(),
100    })?;
101
102    for (entry, metadata) in entries {
103        if metadata.is_symlink() {
104            let path = entry.path();
105            remove_file_if_exists(&path)?;
106
107            if let Some(shim_name) = path.file_stem().and_then(OsStr::to_str) {
108                volta_core::shim::create(shim_name)?;
109            }
110        }
111    }
112
113    Ok(())
114}

116/// Migrate Windows shared directory to use junctions rather than directory symlinks. Similar to
117/// the shims, we previously used symlinks to create the shared global package directory, which
118/// requires developer mode. By using junctions, we can avoid that requirement entirely.
119///
120/// To migrate the directories, we read the shim directory, determine the target of each symlink,
121/// delete the link, and then create a junction (using volta_core::fs::symlink_dir which delegates
122/// to `junction` internally)
123#[cfg(windows)]
124fn migrate_shared_directory(new_home: &v4::VoltaHome) -> Fallible<()> {
125    use std::fs::read_link;
126    use volta_core::fs::{remove_dir_if_exists, symlink_dir};
127
128    let entries =
129        read_dir_eager(new_home.shared_lib_root()).with_context(|| ErrorKind::ReadDirError {
130            dir: new_home.shared_lib_root().to_owned(),
131        })?;
132
133    for (entry, metadata) in entries {
134        if metadata.is_symlink() {
135            let path = entry.path();
136            let source = read_link(&path).with_context(|| ErrorKind::ReadDirError {
137                dir: new_home.shared_lib_root().to_owned(),
138            })?;
139
140            remove_dir_if_exists(&path)?;
141            symlink_dir(source, path).with_context(|| ErrorKind::CreateSharedLinkError {
142                name: entry.file_name().to_string_lossy().to_string(),
143            })?;
144        }
145    }
146
147    Ok(())
